# Call sites redirected into the CFW blob. tools/build_blob.sh turns these into
# rknano_fw.py patches (out/gen/cfw_hooks.patch).
#
#   MOD ADDR OP   SYMBOL         [TAG]
#   bl    replace the instruction with "bl SYMBOL"
#   b     replace it with "b.w SYMBOL"
#   wrap  site must be a bl; it becomes "bl SYMBOL" and the original target is
#         listed in out/gen/wraps.h as WRAP(return address, target, "TAG")
#
# Blob code exists only during a library refresh: only hook mod39/mod40 (and
# mod05's refresh driver, see patches/recent_albums_10_mod05_driver.S).

39 0307b086 bl   cfw_hook_file                   # song record
